Core DAX Measures (powerbi/dax_measures.dax) 
Total Revenue = SUM(order_items[item_total])
Gross Profit = [Total Revenue] - SUMX(order_items, order_items[quantity] * RELATED(products[cost_price]))
MoM Growth % = DIVIDE([Total Revenue] - CALCULATE([Total Revenue], DATEADD(orders[order_date], -1, MONTH)), CALCULATE([Total Revenue], DATEADD(orders[order_date], -1, MONTH)), 0) * 100
Net Savings = CALCULATE(SUM(transactions[amount_usd]), transactions[is_expense] = FALSE()) - CALCULATE(SUM(transactions[amount_usd]), transactions[is_expense] = TRUE())
